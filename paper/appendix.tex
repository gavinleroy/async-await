  \textbf{(Swift)} Modern Swift development involves a lot of asynchronous (or ``async'') programming using closures and completion handlers, but these APIs are hard to use. \subsection{Additional Implementations of timeout}
\label{app:timeout-impls}

\textbf{\Js}

\begin{lstlisting}[language=Javascript]
function sleepCancellable(duration: number, opts?: { signal?: AbortSignal }): Promise<null> {
  let timerId: number;
  let signalHandler: () => void;
  const timer = new Promise<null>((accept, reject) => {
    signalHandler = () => reject(opts?.signal?.reason);
    opts?.signal?.addEventListener("abort", signalHandler, { once: true });
    timerId = setTimeout(() => accept(null), duration);
  })

  return timer.finally(() => {
    clearTimeout(timerId);
    opts?.signal?.removeEventListener("abort", signalHandler);
  });
}

async function timeoutCancellable<T>(  
  duration: number,
  controller: AbortController,
  f: (signal: AbortSignal) => Promise<T>,    
): Promise<T | null> {
  const result = await Promise.race([
    sleepCancellable(duration, { signal: controller.signal }), 
    f(controller.signal)
  ]);
  if (result === null) controller.abort();  
  return result;
}
\end{lstlisting}

\textbf{Rust+Smol}

\begin{lstlisting}[language=Rust]
async fn timeout_task<T, Fut>(duration: Duration, task: Fut) -> Option<T>
where
  T: Send + 'static,
  Fut: Future<Output = T> + Send + 'static,
{
  let mut handle = smol::spawn(task);
  let result = (&mut handle)
    .map(Some)
    .or(sleep(duration).map(|_| None))
    .await;
  if result.is_none() {
    handle.cancel().await;
  }
  result
}
\end{lstlisting}

\subsection{\Eagerness Programs}

\begin{figure}[th]

async fn work(msg: &str) {
  print!("{msg}");

#[tokio::main]
async fn main() {

  let b = work("B");

\begin{lstlisting}[language=Swift]
func work(msg: String
) async throws {
  print(msg)
  try await Task.sleep(
    for: .milliseconds(1))
}

async let a = work(msg: "A")
async let b = work(msg: "B")
print("C")
try await a; try await b
// => CAB, CBA, BCA, ACB, ...
\end{lstlisting}
\cprotect\subcaption{Swift function applications are semi-eager. The right-hand-side of an async let binding is evaluated in a concurrently executing \task, execution of the current thread does not transfer to the body of the applied function. \subsection{\Destruction Programs}

\begin{figure}[H]
  \begin{subfigure}[t]{0.30\textwidth}
\begin{lstlisting}[language=JavaScript,escapechar=|]
async function inner() {
  await new Promise(r =>
    setTimeout(r, 10_000))
  console.log("done")
}

async function work() {
  let a = inner()
  await new Promise(r =>
    setTimeout(r, 0))
  console.log("exiting")
}

await work() // ~10s
\end{lstlisting}
    \cprotect\caption{JavaScript awaits all running tasks at the end of their scope (program scope). The execution of the above program takes about 10 seconds and prints ``exiting,'' and ``done,'' because the leaked \task is fully awaited.}
    % \label{fig:designdims:destruction:await}
  \end{subfigure}\hfill%
  \begin{subfigure}[t]{0.32\textwidth}
    \begin{lstlisting}[language=Swift,escapechar=|]
func inner() async {
  try? await Task.sleep(
      for: .seconds(10))
  if Task.isCancelled {
    print("cancelled")
  }
  print("done")
}

func work() async throws {
  async let _ = inner()
  try await Task.sleep(
      for: .seconds(0))
  print("exiting")
}

try await work() // ~0.5s
\end{lstlisting}
    \cprotect\caption{Swift cancels all running tasks on destruction. The execution of the above program takes about 0.5 seconds and prints ``exiting,'' ``cancelled,'' and ``done.''}
