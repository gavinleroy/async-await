% --- Define Table Macros ---
\newcommand{\DimSection}[1]{%
  \multicolumn{4}{@{}l}{\Large #1} \\ \addlinespace[1ex]
}

\newcommand{\DimRowTitle}[2]{%
  #1 \newline {\footnotesize #2\par}
}

\newcommand{\DimCell}[3]{%
  \textsc{#1} \newline {\footnotesize #2\par\vspace{1ex}#3\par}
}

\newcommand{\DimSeparator}{%
  \arrayrulecolor{gray}\specialrule{1pt}{1ex}{1.5ex}\arrayrulecolor{black}
}

\DimSection{Start of Life}

\DimRowTitle{Eagerness}{How to evaluate an async function application.} &
\DimCell{lazy}{Evaluate to a coroutine without executing further.}{Python, Rust} &
\DimCell{eager}{Evaluate in current thread, and schedule as task on await.}{\CSharp, \Js} &
\DimCell{semi-eager}{Evaluate to task, and schedule task for execution.}{Swift} \\
\addlinespace[1ex]

\DimRowTitle{Suspension}{Guarantees on whether await points suspend.} &
\DimCell{static}{Await points guaranteed to suspend.}{\Js} &
\DimCell{dynamic}{No guarantees on awaiting tasks.}{\CSharp, Swift, Tokio, Smol, Asyncio, Trio} & \\

\DimSeparator

\DimSection{End of Life}

\DimRowTitle{Extent}{The default interval of time during which a task may exist.} &
\DimCell{indefinite}{Tasks by default may exist until the end of the runtime.}{\Js, \CSharp, Tokio, Smol, Asyncio} &
\DimCell{dynamic}{Tasks by default may exist until the end of their spawning scope.}{Swift, Trio} & \\
\addlinespace[1ex]

\DimRowTitle{Reference Strength}{[For indefinite extent] \newline The type of reference to a task held by the runtime.} &
\DimCell{strong}{The runtime holds a strong reference.}{\Js, \CSharp, Tokio} &
\DimCell{weak}{The runtime holds a weak reference.}{Asyncio, Smol} & \\
\addlinespace[1ex]

\DimRowTitle{Destruction}{How a task is cleaned up at the end of its extent.} &
\DimCell{awaited}{The task is awaited to completion.}{\Js, Trio} &
\DimCell{cancelled}{The task is cancelled, and then possibly awaited.}{Swift, Tokio, Smol, Asyncio} &
\DimCell{terminated}{The program exits.}{\CSharp} \\
\addlinespace[1ex]

\DimRowTitle{Propagation}{What happens to exceptions in unawaited tasks.} &
\DimCell{destruction}{Exceptions are reraised by dependents.}{Trio} &
\DimCell{never}{The exception is kept within the task.}{\Js, \CSharp, Tokio, Smol, Asyncio, Swift} & \\

\DimSeparator

\DimSection{Cancellation}

\DimRowTitle{Awareness}{Whether a task is able to respond to being cancelled.} &
\DimCell{unaware}{The task cannot respond to being cancelled.}{Rust} &
\DimCell{aware}{The task can respond to being cancelled.}{Asyncio, Trio, Swift} & \\
\addlinespace[1ex]

\DimRowTitle{Direction}{How cancellation is communicated through the task graph.} &
\DimCell{top-down}{Starting from the root task, and communicated from dependents to dependencies.}{Rust} &
\DimCell{bottom-up}{Starting from the root's dependencies and communicated to dependents.}{Asyncio, Trio} &
\DimCell{simultaneous}{To all transitive dependencies at once.}{Swift} \\
\addlinespace[1ex]

\DimRowTitle{Persistence}{[For aware cancellation] \newline How long a cancellation of a task lasts.} &
\DimCell{transient}{A task can ignore cancellation and proceed as normal.}{Asyncio} &
\DimCell{persistent}{A task can ignore cancellation but remains cancelled.}{Trio, Swift} & \\
