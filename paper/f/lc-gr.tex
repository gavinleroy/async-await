$$
\begin{array}{r@{\ }c@{\ }l}
\mathsf{Const}\ c      & ::=  & \mathsf{true} \mid \mathsf{false} \mid \mathsf{number} \mid \mathsf{string} \\
\mathsf{Expr}\ e       & ::=  & x \mid v \mid \mathbfsf{fn}\ \many{x} \to e \mid e\ \many{e} \\
                       & \mid & \mathsf{let}\ x = e\ \mathsf{in}\ e \\
                       & \mid & \mathsf{if}\ e\ \mathsf{then}\ e\ \mathsf{else}\ e \\
                       & \mid & \mathsf{fix}\ e \mid e \mathbin{;} \dots \mathbin{;} e \\
                       & \mid & \mathsf{reset}\ e \mid \mathsf{shift}\ x \to e \\
                       & \mid & \mathsf{ref}\ e \mid !e \mid e := e \\
\\
\mathsf{Value}\ v      & ::=  & c \mid () \mid \mathsf{ptr}(x) \\
       & \mid & \mathbfsf{fn}\ \many{x} \to e \mid \mathsf{fix}\ v \\
\\
\mathsf{Eval\ Ctx}\ E      & ::=  & [\cdot] \mid \many{v}\ E\ \many{e} \mid \mathsf{fix}\ E \\
       & \mid & \mathsf{let}\ x = E\ \mathsf{in}\ e \\
       & \mid & \mathsf{let}\ x = v\ \mathsf{in}\ E \\
       & \mid & \mathsf{if}\ E\ \mathsf{then}\ e\ \mathsf{else}\ e \\
       & \mid & \many{v} \mathbin{;} E \mathbin{;} \many{e} \mid \mathsf{reset}\ E \\
       & \mid & \mathsf{ref}\ E \mid !E \mid E := e \\
       & \mid & v := E
\\
\mathsf{Delim\ Ctx}\ M      & ::=\  & \dots\ \mathsf{(} E\ \mathsf{without}\ \mathsf{reset}\ M \mathsf{)}
\\
\mathsf{Heap}\ \sigma & ::=  & \cdot \mid \sigma, x \mapsto v
\end{array}
$$
