$$
\begin{array}{r@{\ }c@{\ }l}
c      & ::=  & \text{true} \mid \text{false} \mid \text{number} \mid \text{string}
\\
e      & ::=  & x \mid v \mid \text{fun } \many{x} \to e \mid e\ \many{e} \\
       & \mid & \text{let } x = e \text{ in } e \\
       & \mid & \text{if } e \text{ then } e \text{ else } e \\
       & \mid & \text{fix } e \mid e \mathbin{;} \dots \mathbin{;} e \\
       & \mid & \text{reset } e \mid \text{shift } x \to e \\
       & \mid & \text{ref } e \mid !e \mid e := e \\
\\
v      & ::=  & c \mid () \mid \text{ptr}(\ell) \\
       & \mid & \text{fun } \many{x} \to e \mid \text{fix } v \\
\\
E      & ::=  & [\cdot] \mid \many{v}\ E\ \many{e} \mid \text{fix } E \\
       & \mid & \text{let } x = E \text{ in } e \\
       & \mid & \text{let } x = v \text{ in } E \\
       & \mid & \text{if } E \text{ then } e \text{ else } e \\
       & \mid & \many{v} \mathbin{;} E \mathbin{;} \many{e} \mid \text{reset } E \\
       & \mid & \text{ref } E \mid !E \mid E := e \mid v := E \\
\\
M      & ::=  & \dots \text{ (} E \text{ without } \text{reset } M \text{)}
\\
\Sigma & ::=  & \cdot \mid \Sigma, \ell \mapsto v
\end{array}
$$
