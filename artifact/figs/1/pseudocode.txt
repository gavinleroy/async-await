async fn write_to_log():
  print("A")
  # simulate log write
  await sleep(2)
  print("B")

async fn process_await():
  task = spawn(write_to_log())
  await sleep(0)
  await task

async fn process_detached():
  task = spawn(write_to_log())

async fn ex1():
  t = write_to_log()
  print("C")
  await sleep(3)

async fn ex2():
  await process_detached()
  await sleep(1)
  print("C")

async fn ex3():
  await timeout(1, process_await)
  print("C")
  await sleep(3)
